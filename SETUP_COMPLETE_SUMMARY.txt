━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   ✅ SAFE SESSION INITIALIZATION
      if (session_status() === PHP_SESSION_NONE) {
          session_start();
      }
      └─ Only starts session if not already started
      └─ Prevents "session already active" warnings

   ✅ AUTO TIMEOUT (30 minutes)
      $_SESSION['last_activity'] timestamp tracking
      └─ Auto-logout after 30 minutes of inactivity
      └─ Redirect to login with "session expired" message

   ✅ SESSION VARIABLES
      $_SESSION['user_id']         - Unique user ID
      $_SESSION['email']           - User email address
      $_SESSION['fullname']        - Full name of user
      $_SESSION['role']            - 'admin' or 'user'
      $_SESSION['last_activity']   - Last activity timestamp

   ✅ SECURE ACCESS CONTROL
      requireAdmin()               - Must be admin
      requireLogin()               - Must be logged in
      Role-based redirection       - Redirect to correct dashboard
